// ---- الأرض الطرية قرب الماء ----
let wts = [1, 0.7, 0.4], lastKey = '', warnT = 0;
function soft(dt) {
  const rp = I.rigPos;
  if (I.st.mounted) {
    const key = rp.x + ',' + rp.z;
    if (key !== lastKey) { lastKey = key; wts = [0, 1, 2].map(() => 0.3 + Math.random() * 0.7); warnT = 0; }
    const w = wetness(rp.x, rp.z);
    if (w > 0.3) {
      const k = 0.00004 * w * dt;                     // ≈ 2.4 مم/دقيقة عند الرطوبة الكاملة
      for (let i = 0; i < 3; i++) I.st.leg[i] = Math.max(-0.06, I.st.leg[i] - k * wts[i]);
      warnT += k;
      if (warnT > 0.0015) { warnT = -0.004; N().toast && N().toast('⚠ الأرض طرية قرب الماء: الحامل يهبط ببطء — أعد التسوية قبل كل قراءة'); }
    }
  }
  const g = holder.group.position;
  if (!holder.walking && holder.station) {
    const w = wetness(g.x, g.z);
    if (w > 0.3) g.y = Math.max(groundH(g.x, g.z) - 0.03, g.y - 0.00002 * w * dt);   // تغوص الميرا قليلاً
  }
}

  const dt = Math.min(0.1, (now - last) / 1000); last = now;
  if (T.rock > 0) {
    T.rock -= dt; T.t += dt; apply();
    const s = N().sight && N().sight();
    if (s && s.hit && Number.isFinite(s.reading)) T.min = Math.min(T.min, s.reading);
    if (T.rock <= 0) {

      if (N().say) N().say(ex ? 'انتهى التأرجح: القراءة الصحيحة هي أصغر قيمة رأيتها.' : (T.min < 1e8 ? `أصغر قراءة خلال التأرجح = ${T.min.toFixed(3)} م (هي القراءة الصحيحة)` : 'انتهى التأرجح'), 'ok');
    }
  }
  drawBub(); soft(dt);
  requestAnimationFrame(tick);
})(last);

function randomJob(sd) {
  const r = mulberry32(sd * 131 + 5), E0 = 1000 + Math.round(r() * 8000), N0 = 1000 + Math.round(r() * 8000);
  const wet = (a, b) => { for (let t = 0; t <= 1; t += 0.04) if (isWater(a.x + (b.x - a.x) * t, a.z + (b.z - a.z) * t)) return true; return false; };
  const hs = houseList(), minD = r() < 0.7 ? 30 : 6;   // غالباً بعيد عن النهر، وأحياناً قريب منه
  const okS = (x, z) => {
    if (isWater(x, z) || inHouse(x, z)) return false;
    const ri = riverInfo(x, z);
    return ri.d > ri.hw + minD && hs.every(h => Math.hypot(x - h.x, z - h.z) > h.w + 10);
  };
  let s0 = null;
  for (let k = 0; k < 500 && !s0; k++) {
    const a = r() * 6.283, d = Math.sqrt(r()) * 70, x = Math.round(Math.cos(a) * d), z = Math.round(Math.sin(a) * d);
    if (okS(x, z)) s0 = { x, z };
  }
  if (!s0) s0 = dryNear(0, 0);
  for (let tries = 0; tries < 80; tries++) {
    const pts = [], placed = [{ x: s0.x, z: s0.z }];
    const ok = (x, z) => Math.hypot(x - s0.x, z - s0.z) > 12 && Math.abs(x) < 135 && Math.abs(z) < 135 && !isWater(x, z) && !inHouse(x, z) && placed.every(q => Math.hypot(q.x - x, q.z - z) >= 18) && !wet(s0, { x, z });
    const mk = (id, bm, dmin, dmax) => {
      for (let k = 0; k < 40; k++) {
        const a = r() * 6.283, d = dmin + r() * (dmax - dmin), x = Math.round(s0.x + Math.cos(a) * d), z = Math.round(s0.z + Math.sin(a) * d);
        if (ok(x, z)) { placed.push({ x, z }); const p = { id, x, z, bm, elev: null }; pts.push(p); return p; }
      }
      return null;
    };
    const bm = mk('BM1', true, 25, 45); if (!bm) continue;
    const n = 3 + Math.floor(r() * 3); let good = true;
    for (let i = 1; i <= n; i++) if (!mk('P' + i, false, 20, 85)) good = false;
    if (!good) continue;
    const hi = groundH(s0.x, s0.z) + 1.6, inR = p => { const Rr = hi - groundH(p.x, p.z); return Rr > 0.4 && Rr < 3.6; };
    if (!inR(bm) || pts.filter(p => !p.bm && inR(p)).length < 2) continue;
    bm.elev = Math.round((50 + r() * 450) * 1000) / 1000;
    return { points: pts, setups: [{ id: 'S1', x: s0.x, z: s0.z }], cur: 0, E0, N0 };
  }
  const a = dryNear(s0.x + 30, s0.z - 20), b = dryNear(s0.x - 25, s0.z - 30), c = dryNear(s0.x + 40, s0.z + 35);
  return { points: [{ id: 'BM1', x: a.x, z: a.z, bm: true, elev: 100 }, { id: 'P1', x: b.x, z: b.z, bm: false, elev: null }, { id: 'P2', x: c.x, z: c.z, bm: false, elev: null }], setups: [{ id: 'S1', x: s0.x, z: s0.z }], cur: 0, E0, N0 };
}

function camTargets() {

}

function mapTap(x, z) {
  x = Math.round(clamp(x, -(HALF - 4), HALF - 4)); z = Math.round(clamp(z, -(HALF - 4), HALF - 4));
  if (mapTool === 'pt' || mapTool === 'bm') {
    if (isWater(x, z)) return toast('💧 لا يمكن وضع نقطة داخل الماء');
    if (nearOf(job.points, x, z, 4)) return toast('قريبة جداً من نقطة أخرى');
    const bm = mapTool === 'bm', p = { id: nextId(bm ? 'BM' : 'P'), x, z, bm, elev: null };
    if (bm) { const t = trueRL({ x, z }); p.elev = t === null ? 100 : Math.round(t * 1000) / 1000; }
    job.points.push(p); onJobChanged();
  } else if (mapTool === 'inst') {
    if (isWater(x, z)) return toast('💧 لا يمكن وضع الجهاز داخل الماء (ضعه على الضفة أو الجسر)');
    if (inHouse(x, z)) return toast('🏠 لا يمكن وضع الجهاز داخل مبنى');
    const s0 = nearOf(job.setups, x, z, 6);
    if (s0) gotoSetup(s0);
    else { const s = { id: nextSetup(), x, z }; job.setups.push(s); gotoSetup(s); }
  } else if (mapTool === 'staff') {
    const p = nearOf(job.points, x, z, 10); if (p) sendStaff(p); else toast('المس قرب نقطة لإرسال الهدف إليها');
  } else if (mapTool === 'del') {
    const t = nearAny(x, z, 6);
    if (!t) toast('لا توجد نقطة هنا'); else if ('bm' in t) delPoint(t); else delSetup(t);
  } else if (mapTool === 'pin') {
    const t = nearAny(x, z, 8);
    if (!t) return toast('المس قرب نقطة أو موضع جهاز');
    t.locked = !t.locked; toast(t.locked ? '📌 ثُبّتت ' + t.id : 'أُلغي تثبيت ' + t.id); onJobChanged();
  }
}

  scopeCam.getWorldPosition(_o); scopeCam.getWorldDirection(_d);

  if (!h) return s;
  const lp = g.worldToLocal(_p.copy(h.point));
  s.hit = true; const rdg = pt => face.worldToLocal(_t.copy(pt)).y + 2; s.reading = ts ? h.point.y - gy : rdg(h.point); s.lat = lp.x; s.sd = h.distance; s.alpha = Math.asin(clamp(_d.y, -1, 1));
  s.edm = ts && Math.abs(lp.x) < 0.07 && Math.abs(lp.y - getPrismHeight()) < 0.07;
  if (!ts) {
    _r.set(1, 0, 0).applyQuaternion(scopeCam.quaternion);
    const sp = I.type === 'tach' ? 1 - _d.y * _d.y : 1, a = Math.atan(0.005 * sp);

function fbCompute() {
  let HI = null; const rlMap = {}, out = [];
  rows.forEach(r => {
    const p = ptById(r.pt), known = p && p.bm && p.elev != null ? p.elev : null;
    const sight = r.fs != null ? r.fs : r.is != null ? r.is : null;
    const o = { rl: null, hi: null, calc: null, miscl: null, tru: p ? trueRL(p) : null, err: null };
    if (HI !== null && sight !== null) o.calc = HI - sight;
    if (known !== null) { o.rl = known; if (o.calc !== null) o.miscl = (o.calc - known) * 1000; }
    else if (o.calc !== null) o.rl = o.calc;
    else if (r.pt && rlMap[r.pt] !== undefined) o.rl = rlMap[r.pt];
    if (o.rl !== null && r.pt) rlMap[r.pt] = o.rl;
    if (r.bs != null && o.rl !== null) HI = o.rl + r.bs;
    o.hi = HI;
    if (known !== null) o.err = o.miscl; else if (o.rl !== null && o.tru !== null && o.calc !== null) o.err = (o.rl - o.tru) * 1000;
    out.push(o);
  });
  return out;
}

function fbSummary(comp) {
  let F = -1, L = -1;
  rows.forEach((r, i) => {
    const p = ptById(r.pt);
    if (F < 0 && r.bs != null && comp[i].rl !== null && p && p.bm) F = i;
    if (r.fs != null && comp[i].calc !== null) L = i;
  });
  const lines = [];
  if (F < 0 || L < F) { lines.push('أضف BS على نقطة BM ثم FS على نقطة لاحقة لتظهر فحوص الحساب والإغلاق.'); return lines; }
  let sb = 0, sf = 0, dist = 0;
  for (let i = 0; i <= L; i++) {
    const r = rows[i];
    if (r.bs != null && i !== L) sb += r.bs;
    if (r.fs != null) sf += r.fs;
    if (r.d) { if (i !== L && r.d.bs) dist += r.d.bs; if (r.d.fs) dist += r.d.fs; }
  }
  const dr = comp[L].calc - comp[F].rl, chk = (sb - sf) - dr;
  lines.push(`ΣBS = ${sb.toFixed(3)} | ΣFS = ${sf.toFixed(3)} | ΣBS−ΣFS = ${(sb - sf).toFixed(3)} | ΔRL = ${dr.toFixed(3)}`);
  lines.push(Math.abs(chk) < 0.0006 ? '✓ فحص الحساب سليم' : `✗ فحص الحساب غير مطابق (فرق ${(chk * 1000).toFixed(1)} مم) — راجع الجمع والطرح`);
  if (comp[L].miscl !== null) {
    const allow = dist > 0 ? 12 * Math.sqrt(dist / 1000) : null, m = comp[L].miscl;
    lines.push(`خطأ الإغلاق = ${m >= 0 ? '+' : ''}${m.toFixed(1)} مم` + (allow !== null ? ` | المسموح 12√K = ${allow.toFixed(1)} مم (K = ${(dist / 1000).toFixed(3)} كم) → ${Math.abs(m) <= allow ? '✓ مقبول' : '✗ مرفوض'}` : ''));
  }
  return lines;
}

function ttCalc(r) {
  const su = setupById(r.su), p = ptById(r.pt), o = {};
  if (!su || !p) return o;
  const zen = SM.zenith(r.v, r.face), Zs = zOf(su);
  o.Dt = Math.hypot(p.x - su.x, p.z - su.z); o.Zt = zOf(p);
  if (r.k === 'tach' || r.k === 'theo') {
    if (r.u != null && r.l != null) {
      Object.assign(o, r.k === 'tach' ? SM.stadiaReduced(r, Zs) : SM.stadiaClassic(r, zen, Zs));
      o.eD = (o.D - o.Dt) * 100; if (o.Z != null) o.eZ = (o.Z - o.Zt) * 1000;
    }
  } else if (r.sd != null) {
    const c0 = crd(su.x, su.z), ct = crd(p.x, p.z);
    Object.assign(o, SM.totalStation(r, c0, Zs));
    o.eN = (o.N - ct.N) * 1000; o.eE = (o.E - ct.E) * 1000; o.eZ = (o.Z - o.Zt) * 1000;
  }
  return o;
}

function ttSummary(reveal) {
  const rr = rowsT.filter(r => r.k === I.type), lines = [], show = reveal || !CFG.exam;
  if (!rr.length) return ['لا توجد قياسات بعد. ' + (I.type === 'ts' ? 'وجّه الشعيرة إلى المنشور واضغط 📏 قياس وحفظ.' : 'اقرأ ق ع/ق و/ق س (U/M/L) واضغط ➕ تسجيل.')];
  lines.push('عدد القياسات: ' + rr.length);
  if (isStad() && rr.length >= 2) {
    const a = rr[rr.length - 2], b = rr[rr.length - 1], su = setupById(a.su), pa = ptById(a.pt), pb = ptById(b.pt);
    if (a.su === b.su && su && pa && pb && a.pt !== b.pt && a.face === b.face) {
      const ang = (((b.hz - a.hz) % TAU) + TAU) % TAU, bt = p => Math.atan2(p.x - su.x, -(p.z - su.z)), trAng = (((bt(pb) - bt(pa)) % TAU) + TAU) % TAU;
      lines.push(`الزاوية الأفقية ${a.pt}→${b.pt} = ${dms(ang, 1)}` + (show ? ` | الحقيقية ${dms(trAng, 1)} | الخطأ ${(wrapPi(ang - trAng) / SEC).toFixed(0)}″` : ''));
    }
  }
  const by = {};
  rr.forEach(r => { const k = r.su + '|' + r.pt; (by[k] = by[k] || {})[r.face] = r; });
  Object.values(by).forEach(g => {
    if (!g[1] || !g[2]) return;
    const a = g[1], b = g[2], ie = (a.v + b.v - TAU) / 2, hzd = wrapPi(b.hz - a.hz - Math.PI);
    const zz = Math.min(Math.max(a.v, 0.2), Math.PI - 0.2), c2 = hzd * Math.sin(zz) / 2, mh = a.hz + wrapPi(b.hz - Math.PI - a.hz) / 2;
    let s = `${a.pt} (الوجهان): Hz متوسط = ${dms(mh, 1)} | i المقدّر = ${(wrapPi(ie) / SEC).toFixed(0)}″ | |c| المقدّر = ${(Math.abs(c2) / SEC).toFixed(0)}″`;
    if (show && I.ERR.on) s += ` | الحقيقي: i = ${I.errArc('i').toFixed(0)}″، |c| = ${Math.abs(I.errArc('c')).toFixed(0)}″` + (I.type === 'tach' ? ' (i يشمل برغي الفهرس)' : '');
    lines.push(s);
  });
  if (show) {
    const es = [];
    rr.forEach(r => { const o = ttCalc(r); if (isStad()) { if (o.eZ != null) es.push(Math.abs(o.eZ)); } else if (o.eN != null) es.push(Math.max(Math.abs(o.eN), Math.abs(o.eE), Math.abs(o.eZ))); });
    if (es.length) lines.push(`أكبر خطأ في المنسوب/الإحداثيات = ${Math.max(...es).toFixed(0)} مم`);
  }
  return lines;
}

const VERT = `
varying vec2 vUv;
void main(){ vUv = uv; gl_Position = vec4(position.xy, 0.0, 1.0); }`;

const FRAG = `
uniform sampler2D tMap; uniform float uBlur; uniform float uSoft; uniform float uMode; uniform float uSpread; varying vec2 vUv;
float ln(float d, float w, float s){ return 1.0 - smoothstep(w, w + s, abs(d)); }
float segd(vec2 p, vec2 a, vec2 b){ vec2 pa = p - a, ba = b - a; float h = clamp(dot(pa, ba) / dot(ba, ba), 0.0, 1.0); return length(pa - ba * h); }
void main(){
  vec2 p = vUv * 2.0 - 1.0;
  float r = length(p);
  vec3 col;
  if (uBlur < 0.0015) { col = texture2D(tMap, vUv).rgb; }
  else {
    float bias = log2(1.0 + uBlur * 700.0);
    float o = uBlur * 0.4;
    col = vec3(0.0);
    for (int i = -1; i <= 1; i++) { for (int j = -1; j <= 1; j++) { col += texture2D(tMap, vUv + vec2(float(i), float(j)) * o, bias).rgb; } }
    col /= 9.0;
  }
  float s = uSoft, w = 0.0045, ws = 0.003 + s;
  // uMode: 0 = تيودوليت ، 1 = تاكيومتر يدوي مختزل (4 شعيرات أفقية + قطعة مائلة) ، 2 = آلي
  float m0 = step(uMode, 0.5), m1 = step(0.5, uMode) * step(uMode, 1.5), m2 = step(1.5, uMode);
  float full = m0 + m2;
  float hy = 0.4775 * uSpread;
  float h1 = ln(p.y, w, ws) * full;
  float v1 = ln(p.x, w, ws);
  float sd = (ln(p.y - hy, w, ws) + ln(p.y + hy, w, ws)) * step(abs(p.x), 0.24) * m0;
  float tU = ln(p.y - hy, w, ws) * step(abs(p.x), 0.32);      // ق ع
  float tM = ln(p.y, w, ws) * step(abs(p.x), 0.3);            // ق و
  float tS = ln(p.y + 0.167, w, ws) * step(abs(p.x), 0.3);    // S (شعيرة التصويب)
  float tL = ln(p.y + hy, w, ws) * step(abs(p.x), 0.32);      // ق س
  float tI = 1.0 - smoothstep(w, w + ws, segd(p, vec2(0.32, hy), vec2(-0.38, 0.17 * hy))); // قطعة مائلة تخرج من العليا
  float line = clamp(h1 + v1 + sd + (tU + tM + tS + tL + tI) * m1, 0.0, 1.0);
  float halo = clamp(ln(p.y, w * 2.6, 0.004 + s) * full + ln(p.x, w * 2.6, 0.004 + s), 0.0, 1.0) * 0.45;
  col = mix(col, vec3(1.0), halo);
  col = mix(col, vec3(0.02), line * (1.0 - clamp(s * 10.0, 0.0, 0.6)));
  col = mix(vec3(0.02, 0.02, 0.03), col, 1.0 - smoothstep(0.985, 1.0, r));
  gl_FragColor = vec4(col, 1.0);
  #include <colorspace_fragment>
}`;

export function createScopeFX(renderer) {
  const rt = new THREE.WebGLRenderTarget(512, 512, { minFilter: THREE.LinearMipmapLinearFilter, magFilter: THREE.LinearFilter, generateMipmaps: true });
  const mat = new THREE.ShaderMaterial({
    uniforms: { tMap: { value: rt.texture }, uBlur: { value: 0 }, uSoft: { value: 0 }, uMode: { value: 0 }, uSpread: { value: 1 } },
    vertexShader: VERT, fragmentShader: FRAG, depthTest: false, depthWrite: false
  });
  const qs = new THREE.Scene(), quad = new THREE.Mesh(new THREE.PlaneGeometry(2, 2), mat);
  quad.frustumCulled = false; qs.add(quad);
  const qc = new THREE.OrthographicCamera(-1, 1, 1, -1, 0, 1);
  let size = 512;
  return {
    setMode(m) { mat.uniforms.uMode.value = m; },
    setStadia(on) { mat.uniforms.uMode.value = on ? 0 : 2; },
    setSpread(v) { mat.uniforms.uSpread.value = v; },
    draw(scene, cam, vp, blur, soft) {
      const px = Math.max(160, Math.min(768, Math.round(vp.w * renderer.getPixelRatio())));
      if (px !== size) { size = px; rt.setSize(px, px); }
      renderer.setScissorTest(false);
      renderer.setRenderTarget(rt); renderer.render(scene, cam); renderer.setRenderTarget(null);
      mat.uniforms.uBlur.value = blur; mat.uniforms.uSoft.value = soft;
      renderer.setScissorTest(true);
      renderer.setViewport(vp.x, vp.y, vp.w, vp.h); renderer.setScissor(vp.x, vp.y, vp.w, vp.h);
      renderer.render(qs, qc);
      renderer.setScissorTest(false);
    }
  };
}

// تاكيومتري كلاسيكي (تيودوليت): D = 100·s·cos²α ، ΔZ = i + 50·s·sin2α − M
export function stadiaClassic(r, zen, Zs) {
  const s = r.u - r.l, a = elev(zen), m = r.m != null ? r.m : (r.u + r.l) / 2;
  const D = 100 * s * Math.cos(a) ** 2, dz = (r.i ?? 0) + 50 * s * Math.sin(2 * a) - m;
  return { s, D, dz, Z: Zs + dz };
}
// تاكيومتر ذاتي الاختزال (قوانين المقرر): D = (ق ع − ق س)×100 = (ق ع − ق و)×200 ، ΔH = (ق ع − ق و)·K + J − S
export function stadiaReduced(r, Zs) {
  const mm = r.m != null ? r.m : (r.u + r.l) / 2, ss = r.s != null ? r.s : mm, o = { D: (r.u - r.l) * 100, D2: (r.u - mm) * 200 };
  if (r.kc != null) { o.dH = (r.u - mm) * r.kc + (r.i ?? 0) - ss; o.Z = Zs + o.dH; }
  return o;
}
// محطة شاملة: HD = SD·sinZ ، VD = SD·cosZ ، إحداثيات الهدف
export function totalStation(r, c0, Zs) {
  const zen = zenith(r.v, r.face), az = r.face === 2 ? r.hz - Math.PI : r.hz;
  const HD = r.sd * Math.sin(zen), VD = r.sd * Math.cos(zen);
  return { HD, VD, N: c0.N + HD * Math.cos(az), E: c0.E + HD * Math.sin(az), Z: Zs + (r.i ?? 0) + VD - (r.hr ?? 0) };
}
// قياسان على نقطة واحدة بالوجهين (a وجه I و b وجه II): خطأ الفهرس وانحراف خط النظر والمتوسط
export function twoFace(a, b) {
  const ie = wrapPi((a.v + b.v - TAU) / 2), hzd = wrapPi(b.hz - a.hz - Math.PI), zz = Math.min(Math.max(a.v, 0.2), Math.PI - 0.2);
  return { ie, c: hzd * Math.sin(zz) / 2, meanHz: a.hz + wrapPi(b.hz - Math.PI - a.hz) / 2 };
}

// مقبض مسنّن (knurled)
function knurl(r, h, n, mat) {
  const g = new THREE.Group(); g.add(new THREE.Mesh(cyl(r - 0.0008, h, 24), mat));
  for (let i = 0; i < n; i++) { const a = i / n * TAU, b = mesh(box(0.0016, h, 0.0022), mat, Math.cos(a) * r, 0, Math.sin(a) * r); b.rotation.y = -a; g.add(b); }
  return g;
}
// دعامتان على شكل حرف U مدرّجتان بحواف منحنية
function uBracket(a1, H, gap, yb, D, mat) {
  const s = new THREE.Shape(), a0 = a1 + 0.008, g = gap / 2, r = 0.01;
  s.moveTo(-a0, 0); s.lineTo(a0, 0); s.lineTo(a1, H - r); s.quadraticCurveTo(a1 - 0.002, H, a1 - r - 0.002, H);
  s.lineTo(g + r, H); s.quadraticCurveTo(g, H, g, H - r); s.lineTo(g, yb); s.lineTo(-g, yb); s.lineTo(-g, H - r);
  s.quadraticCurveTo(-g, H, -g - r, H); s.lineTo(-a1 + r + 0.002, H); s.quadraticCurveTo(-a1 + 0.002, H, -a1, H - r); s.lineTo(-a0, 0);
  const geo = new THREE.ExtrudeGeometry(s, { depth: D, bevelEnabled: true, bevelThickness: 0.003, bevelSize: 0.003, bevelSegments: 2, curveSegments: 6 });
  geo.translate(0, 0, -D / 2);
  return new THREE.Mesh(geo, mat);
}

export function update(dt) {
  sanitize();
  st.explode += (st.exTarget - st.explode) * Math.min(1, dt * 4);
  if (Math.abs(st.exTarget - st.explode) < 0.001) st.explode = st.exTarget;
  const hs = headState();
  headG.position.set(0, hs.Hc, 0);
  headG.quaternion.setFromUnitVectors(UP, _v.set(-hs.tx, 1, -hs.tz).normalize());
  tripAnchor.position.set(0.3, hs.Hc * 0.55, 0.45);
  updateLegs();
  headG.getWorldQuaternion(_q);
  _v.set(hs.tx, 0, hs.tz).multiplyScalar(0.18).applyQuaternion(_q.invert());
  let hx = _v.x, hz = _v.z; const hl = Math.hypot(hx, hz);
  if (hl > 0.0065) { hx *= 0.0065 / hl; hz *= 0.0065 / hl; }

  SCREWS.forEach((s, i) => {
    const y0 = hs.Hc + hs.tx * s.x + hs.tz * s.z + 0.008, len = S0 + st.e[i], p = screwParts[i];
    p.home.set(s.x, y0, s.z);

  });

  T = { gx, gz, arc: Math.hypot(gx, gz) * 3437.75 };
  instRoot.position.set(0, sy / 3, 0);
  instRoot.quaternion.setFromUnitVectors(UP, _v.set(-gx, 1, -gz).normalize());

    _f.set(Math.sin(c), Math.cos(ph) * Math.cos(c), -Math.sin(ph) * Math.cos(c));
    const fx0 = _f.x * Math.cos(tz) - _f.y * Math.sin(tz), fy0 = _f.x * Math.sin(tz) + _f.y * Math.cos(tz);
    _f.x = fx0; _f.y = fy0; _f.applyQuaternion(_q2);
    _u.set(0, 1, 0).applyQuaternion(_q2); cam.up.copy(_u);
  } else {
    _f.set(0, 0, -1).applyQuaternion(_q2);
    if (T.arc <= ty.comp) {
      _f.y = 0; _f.normalize();
      const ga = T.gx * Math.sin(st.heading) - T.gz * Math.cos(st.heading);
      const el = ev('c') * SEC + 0.02 * ga, ce = Math.cos(el);
      _f.set(_f.x * ce, Math.sin(el), _f.z * ce); cam.up.set(0, 1, 0);
    } else { _u.set(0, 1, 0).applyQuaternion(_q2); cam.up.copy(_u); }
  }
  _f.add(_o); cam.lookAt(_f);
}

}

export function mulberry32(a) {
  return function () { a |= 0; a = (a + 0x6D2B79F5) | 0; let t = Math.imul(a ^ (a >>> 15), 1 | a); t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t; return ((t ^ (t >>> 14)) >>> 0) / 4294967296; };
}
const clamp = (v, a, b) => Math.min(b, Math.max(a, v));
const smooth = (a, b, x) => { const t = clamp((x - a) / (b - a), 0, 1); return t * t * (3 - 2 * t); };
const hash = (i, j) => { let h = (Math.imul(i, 374761393) + Math.imul(j, 668265263)) | 0; h = Math.imul(h ^ (h >>> 13), 1274126177); return ((h ^ (h >>> 16)) >>> 0) / 4294967295; };
function vnoise(x, z) {
  const xi = Math.floor(x), zi = Math.floor(z), xf = x - xi, zf = z - zi;
  const a = hash(xi, zi), b = hash(xi + 1, zi), c = hash(xi, zi + 1), d = hash(xi + 1, zi + 1);
  const u = xf * xf * (3 - 2 * xf), v = zf * zf * (3 - 2 * zf);
  return a + (b - a) * u + (c - a) * v + (a - b - c + d) * u * v;
}
function fbm(x, z) { let s = 0, a = 0.5, f = 1; for (let i = 0; i < 4; i++) { s += a * vnoise(x * f, z * f); f *= 2; a *= 0.5; } return s; }

export function cylBetween(a, b, r, mat, seg = 8) {
  const d = new V3().subVectors(b, a), L = d.length();
  const m = new THREE.Mesh(new THREE.CylinderGeometry(r, r, L, seg), mat);
  m.position.copy(a).addScaledVector(d, 0.5);
  m.quaternion.setFromUnitVectors(new V3(0, 1, 0), d.normalize());
  return m;
}

  const rr = Math.hypot(x, z);
  const far = smooth(90, 230, rr), rel = 0.5 + 0.5 * far;
  const wx = x + 30 * (vnoise(x * 0.008 + P.ox, z * 0.008 + P.oz) - 0.5), wz = z + 30 * (vnoise(x * 0.008 + P.oz, z * 0.008 + P.ox) - 0.5);
  const big = (fbm(wx * P.f + P.ox, wz * P.f + P.oz) - 0.47) * 2 * P.amp;
  const rg = 1 - Math.abs(2 * vnoise(wx * 0.018 + P.ox * 2.1, wz * 0.018 + P.oz * 1.7) - 1);
  const mid = (rg * rg - 0.3) * P.amp2 * 1.6 + (fbm(x * 0.03 + P.ox * 3, z * 0.03 + P.oz * 3) - 0.47) * 0.35;
  let h = (big + mid) * rel + (P.tx * x + P.tz * z) * (0.35 + 0.65 * far);
  const tz = smooth(0.58, 0.72, vnoise(x * 0.005 + P.ox * 1.9, z * 0.005 + P.oz * 1.1)) * far;
  if (tz > 0) { const s = 1.0, k = h / s, f = k - Math.floor(k); h += ((Math.floor(k) + smooth(0.6, 1.0, f)) * s - h) * 0.6 * tz; }
  const rim = smooth(170, 340, rr);
  return h + rim * rim * P.rim * (0.6 + 0.8 * vnoise(x * 0.01 + 5, z * 0.01 + 9));
}

// ---------- النهر ----------
let RV = null, BRG = null;
const rvCl = s => RV.off + RV.A1 * Math.sin(RV.k1 * s + RV.p1) + RV.A2 * Math.sin(RV.k2 * s + RV.p2);
const rvDcl = s => RV.A1 * RV.k1 * Math.cos(RV.k1 * s + RV.p1) + RV.A2 * RV.k2 * Math.cos(RV.k2 * s + RV.p2);
const rvHW = s => RV.w0 + RV.w1 * (vnoise(s * 0.02 + P.ox * 0.5, P.oz) - 0.5) * 2;
function wlAt(s) {
  const f = (s + 350) / RV.ds, i = Math.max(0, Math.min(RV.wl.length - 2, Math.floor(f))), t = Math.max(0, Math.min(1, f - i));
  return RV.wl[i] * (1 - t) + RV.wl[i + 1] * t;
}
export function riverInfo(x, z) {
  if (!RV) return { d: 1e9, hw: 0, s: 0, wl: 0 };
  const s = x * RV.dx + z * RV.dz, n = x * RV.nx + z * RV.nz, dc = rvDcl(s);
  return { d: Math.abs(n - rvCl(s)) / Math.sqrt(1 + dc * dc), hw: rvHW(s), s, wl: wlAt(s) };
}
function makeRiver() {
  const r = mulberry32(P.seed * 101 + 77), ang = r() * Math.PI;
  RV = { dx: Math.cos(ang), dz: Math.sin(ang), nx: -Math.sin(ang), nz: Math.cos(ang), off: (r() - 0.5) * 110,
    A1: 20 + r() * 20, k1: 0.012 + r() * 0.008, p1: r() * 6.28, A2: 5 + r() * 7, k2: 0.04 + r() * 0.02, p2: r() * 6.28,
    w0: 3.2 + r() * 1.5, w1: 1.2, ds: 5, wl: [] };
  // منسوب الماء: متوسط الأرض الأصلية على المجرى − 0.7 م مع انحدار أحادي الاتجاه (يجري نحو +s)
  const n = 141, raw = [];
  for (let i = 0; i < n; i++) { const s = -350 + i * RV.ds, c = rvCl(s); raw.push(baseH(s * RV.dx + c * RV.nx, s * RV.dz + c * RV.nz)); }
  let prev = Infinity;
  RV.wl = raw.map((_, i) => {
    let a = 0, k = 0; for (let j = Math.max(0, i - 8); j <= Math.min(n - 1, i + 8); j++) { a += raw[j]; k++; }
    prev = Math.min(prev - 0.0007 * RV.ds, a / k - 0.7); return prev;
  });
  makeBridge();
}
// تحفير المجرى: قاع، ضفاف منحدرة، ثم انتقال تدريجي إلى الأرض المحيطة
function carve(h0, x, z) {
  const ri = riverInfo(x, z), d = ri.d, hw = ri.hw, BW = 9, FADE = 14;
  if (d > hw + BW + FADE) return h0;
  const wl = ri.wl;
  let target;
  if (d < hw) { const q = d / hw; target = wl - 0.2 - 0.9 * (1 - q * q); }
  else if (d < hw + BW) target = wl - 0.2 + 1.5 * smooth(0, 1, (d - hw) / BW);
  else target = wl + 1.3 + (d - hw - BW) * 0.02;
  const w = d < hw + BW ? 1 : 1 - smooth(hw + BW, hw + BW + FADE, d);
  return h0 + (target - h0) * w;
}
function terrainH(x, z) { const h0 = baseH(x, z); return RV ? carve(h0, x, z) : h0; }

// الجسر: حيث يقطع الطريق النهر
function makeBridge() {
  BRG = null;
  let best = null;
  for (let z = -230; z <= 230; z += 2) { const x = roadX(z), ri = riverInfo(x, z); if (!best || ri.d < best.d) best = { x, z, d: ri.d }; }
  if (!best || best.d > 10) return;
  const tx = roadX(best.z + 4) - roadX(best.z - 4), tz = 8, tl = Math.hypot(tx, tz), ux = tx / tl, uz = tz / tl;
  const ri = riverInfo(best.x, best.z), dc = rvDcl(ri.s);
  let rx = RV.dx + RV.nx * dc, rz = RV.dz + RV.nz * dc; const rl = Math.hypot(rx, rz); rx /= rl; rz /= rl;
  const cosf = Math.max(0.35, Math.abs(ux * -rz + uz * rx)), hl = (ri.hw + 6) / cosf + 3;
  const e1 = terrainH(best.x + ux * hl, best.z + uz * hl), e2 = terrainH(best.x - ux * hl, best.z - uz * hl);
  BRG = { x: best.x, z: best.z, ux, uz, hl, hw: 2.1, y: Math.max(e1, e2, ri.wl + 1.1) + 0.12 };
}
const onDeck = (x, z) => {
  if (!BRG) return false;
  const dx = x - BRG.x, dz = z - BRG.z;
  return Math.abs(dx * BRG.ux + dz * BRG.uz) <= BRG.hl && Math.abs(-dx * BRG.uz + dz * BRG.ux) <= BRG.hw;
};
// ارتفاع السطح الذي يقف عليه الجهاز/الميرا: الأرض أو سطح الجسر
export function groundH(x, z) {
  if (!P) return 0;

}
export const bridgeInfo = () => BRG;
export function isWater(x, z) { if (!RV) return false; const ri = riverInfo(x, z); return ri.d < ri.hw + 1.8 && !onDeck(x, z); }
const inRiver = (x, z, m = 0) => { if (!RV) return false; const ri = riverInfo(x, z); return ri.d < ri.hw + 1.8 + m; };
export function wetness(x, z) { if (!RV || onDeck(x, z)) return 0; const ri = riverInfo(x, z); return clamp(1 - (ri.d - ri.hw) / 16, 0, 1); }
const localDry = (x, z) => {
  if (!RV) return P.dry;
  const ri = riverInfo(x, z);
  return clamp(P.dry + (ri.d - 50) / 260 - 0.25 * clamp(1 - (ri.d - ri.hw) / 16, 0, 1), 0, 1);
};
export function inHouse(x, z) { return HOUSES.some(h => Math.hypot(x - h.x, z - h.z) < h.rad * 0.7); }

export function riverPath() {
  if (!RV) return [];

  for (let s = -340; s <= 340; s += 6) { const c = rvCl(s); o.push({ x: s * RV.dx + c * RV.nx, z: s * RV.dz + c * RV.nz, hw: rvHW(s) + 1.8, s }); }

}
// أقرب موضع جاف (بعيد 6 م عن الماء وخارج المباني)
export function dryNear(x, z) {
  for (let rad = 0; rad < 140; rad += 3) for (let a = 0; a < 6.283; a += 0.7) {
    const px = Math.round(x + Math.cos(a) * rad), pz = Math.round(z + Math.sin(a) * rad), ri = riverInfo(px, pz);
    if (!isWater(px, pz) && !inHouse(px, pz) && ri.d > ri.hw + 6) return { x: px, z: pz };
  }
  return { x: Math.round(x), z: Math.round(z) };
}

}

// إخفاء ما يحجب خطوط النظر بين مواضع الجهاز والنقاط، وما حول النقاط نفسها
export function clearVeg(points, setups) {
  const pts = points.concat(setups), segs = [], Z0 = new THREE.Matrix4().makeScale(0, 0, 0);
  points.forEach(p => setups.forEach(s => segs.push([s.x, s.z, p.x, p.z])));
  const sd = (x, z, s) => {
    const dx = s[2] - s[0], dz = s[3] - s[1], t = Math.max(0, Math.min(1, ((x - s[0]) * dx + (z - s[1]) * dz) / (dx * dx + dz * dz || 1)));
    return Math.hypot(x - s[0] - t * dx, z - s[1] - t * dz);
  };
  const blocked = (x, z, rad) => pts.some(q => Math.hypot(x - q.x, z - q.z) < rad + 3.5) || segs.some(s => sd(x, z, s) < rad + 2.2);
  const m = new THREE.Matrix4();
  VEG.forEach(o => {
    let dirty = false;
    for (let i = 0; i < o.n; i++) {
      const hide = blocked(o.xz[i * 3], o.xz[i * 3 + 1], o.xz[i * 3 + 2]);
      if (hide !== !!o.hid[i]) { o.hid[i] = hide ? 1 : 0; o.mesh.setMatrixAt(i, hide ? Z0 : m.fromArray(o.base, i * 16)); dirty = true; }
    }
    if (dirty) o.mesh.instanceMatrix.needsUpdate = true;
  });
  HOUSES.forEach(h => { h.g.visible = true; });
}

// ---------- صورة الخريطة (كنتورية) ----------
export function mapImage(half = 120, res = 240) {
  if (mapCache && mapCache.seed === P.seed && mapCache.half === half && mapCache.res === res) return mapCache;
  const n = res + 2, step = 2 * half / res, hs = new Float32Array(n * n);
  let mn = 1e9, mx = -1e9;
  for (let j = 0; j < n; j++) for (let i = 0; i < n; i++) { const h = groundH(-half + (i - 0.5) * step, -half + (j - 0.5) * step); hs[j * n + i] = h; if (h < mn) mn = h; if (h > mx) mx = h; }
  const range = Math.max(0.5, mx - mn), iv = range > 14 ? 2 : range > 7 ? 1 : range > 3.5 ? 0.5 : 0.25;
  const cv = document.createElement('canvas'); cv.width = cv.height = res;
  const ctx = cv.getContext('2d'), img = ctx.createImageData(res, res);
  for (let j = 0; j < res; j++) for (let i = 0; i < res; i++) {
    const k = (j + 1) * n + (i + 1), h = hs[k], t = (h - mn) / range;
    const dx = (hs[k + 1] - hs[k - 1]) / (2 * step), dz = (hs[k + n] - hs[k - n]) / (2 * step);
    const sh = clamp((0.5 * dx + 0.7 + 0.5 * dz) / Math.sqrt(dx * dx + 1 + dz * dz) / 0.99 * 1.2, 0.45, 1.15);
    let r, g, b;
    if (t < 0.5) { const u = t / 0.5; r = 0.38 + 0.47 * u; g = 0.64 + 0.18 * u; b = 0.38 + 0.1 * u; }
    else { const u = (t - 0.5) / 0.5; r = 0.85 - 0.2 * u; g = 0.82 - 0.35 * u; b = 0.48 - 0.15 * u; }
    r *= sh; g *= sh; b *= sh;
    const x = -half + (i + 0.5) * step, z = -half + (j + 0.5) * step;
    if (Math.floor(h / iv) !== Math.floor(hs[k + 1] / iv) || Math.floor(h / iv) !== Math.floor(hs[k + n] / iv)) { r *= 0.55; g *= 0.55; b *= 0.55; }
    if (Math.abs(x - roadX(z)) < 1.6) { r = 0.78; g = 0.66; b = 0.5; }
    { const ri = riverInfo(x, z); if (ri.d < ri.hw + 1.8) { r = 0.16; g = 0.42; b = 0.72; } else if (ri.d < ri.hw + 5) { r *= 0.8; g *= 0.88; } if (onDeck(x, z)) { r = 0.6; g = 0.5; b = 0.38; } }
    const o = (j * res + i) * 4; img.data[o] = r * 255; img.data[o + 1] = g * 255; img.data[o + 2] = b * 255; img.data[o + 3] = 255;
  }
  ctx.putImageData(img, 0, 0);
  mapCache = { canvas: cv, mn, mx, iv, half, res, seed: P.seed };
  return mapCache;
}

// ---------- تدريج الميرا ----------
function staffTexture(pattern, label) {
  const Wd = 144, H = 4000, PW = 76, cv = document.createElement('canvas'); cv.width = Wd; cv.height = H;
  const c = cv.getContext('2d'), Y = h => H - h * 1000;
  c.fillStyle = '#f6f3e6'; c.fillRect(0, 0, Wd, H);
  c.fillStyle = '#161616';
  for (let i = 0; i < 400; i += 2) c.fillRect(0, Y((i + 1) / 100), PW, 10);
  if (pattern === 'E') for (let k = 0; k < 80; k++) c.fillRect(k % 2 === 0 ? 0 : PW - 16, Y((5 * k + 5) / 100), 16, 50);
  else { c.fillStyle = '#c11'; for (let k = 0; k < 80; k++) c.fillRect(0, Y((5 * k + 5) / 100) - 1, PW, 3); }
  c.textAlign = 'center'; c.textBaseline = 'alphabetic';
  const cx = PW + (Wd - PW) / 2;
  for (let n = 0; n < 40; n++) {
    const m = Math.floor(n / 10), d = n % 10, y0 = Y(n / 10);
    c.fillStyle = m % 2 === 1 ? '#d11' : '#111';
    if (label === 'dots') {
      c.font = 'bold 56px sans-serif'; c.fillText(String(d), cx, y0 - 12);
      for (let j = 0; j < m; j++) { c.beginPath(); c.arc(cx + (j - (m - 1) / 2) * 14, y0 - 80, 5, 0, 7); c.fill(); }
    } else if (n > 0 && label === 'pair') { c.font = 'bold 46px sans-serif'; c.fillText(`${m}${d}`, cx, y0 - 30); }
    else if (n > 0 && label === 'stack') { c.font = 'bold 40px sans-serif'; c.fillText(String(m), cx, y0 - 52); c.fillText(String(d), cx, y0 - 10); }
    else if (n > 0 && label === 'dec') { c.font = 'bold 38px sans-serif'; c.fillText(`${m},${d}`, cx, y0 - 30); }
    c.fillStyle = '#666'; c.fillRect(PW, y0 - 1, Wd - PW, 2);
  }
  const t = new THREE.CanvasTexture(cv);
  t.colorSpace = THREE.SRGBColorSpace; t.anisotropy = renderer.capabilities.getMaxAnisotropy();
  return t;
}

export function setMarkers(points, setups) {
  while (markGroup.children.length) {
    const o = markGroup.children.pop();
    if (o.material) { if (o.material.map) o.material.map.dispose(); o.material.dispose(); }
    if (o.geometry) o.geometry.dispose();
  }
  points.forEach(p => {
    const gy = groundH(p.x, p.z);
    const peg = new THREE.Mesh(new THREE.CylinderGeometry(0.03, 0.03, 0.25, 8), new THREE.MeshLambertMaterial({ color: p.bm ? 0x2ea043 : 0xdd2222 }));
    peg.position.set(p.x, gy + 0.06, p.z); markGroup.add(peg);
    const sp = makeLabel(p.id, p.bm ? '#2ea043' : '#e8402a'); sp.position.set(p.x, gy + 5.4, p.z); markGroup.add(sp);
  });
  setups.forEach(s => {

  });
  clearVeg(points, setups);
}

const crossesWater = (x0, z0, x1, z1) => { for (let t = 0; t <= 1; t += 0.04) if (isWater(x0 + (x1 - x0) * t, z0 + (z1 - z0) * t)) return true; return false; };
function planRoute(x0, z0, x1, z1) {
  if (!BRG || !crossesWater(x0, z0, x1, z1)) return [];
  const e = BRG.hl + 1.5, a = { x: BRG.x - BRG.ux * e, z: BRG.z - BRG.uz * e }, b = { x: BRG.x + BRG.ux * e, z: BRG.z + BRG.uz * e };
  return Math.hypot(x0 - a.x, z0 - a.z) < Math.hypot(x0 - b.x, z0 - b.z) ? [a, b] : [b, a];
}
export function sendTo(pt) {
  if (!pt || holder.walking || (holder.station && holder.station.id === pt.id)) return false;
  const g = holder.group.position;
  holder.route = planRoute(g.x, g.z, pt.x, pt.z);   // يعبر النهر عبر الجسر إن وُجد
  holder.target = pt; holder.walking = true; holder.station = null;
  return true;
}
export function updateHolder(dt) {
  if (!holder.walking) return;
  const g = holder.group, wp = holder.route && holder.route.length ? holder.route[0] : null, t = wp || holder.target;
  const dx = t.x - g.position.x, dz = t.z - g.position.z, d = Math.hypot(dx, dz);
  if (d < 0.2) { if (wp) { holder.route.shift(); return; } placeHolderAt(holder.target); return; }
  const k = Math.min(3.2 * dt, d) / d;
  g.position.x += dx * k; g.position.z += dz * k;
  g.position.y = groundH(g.position.x, g.position.z);
  g.rotation.y = Math.atan2(dx, dz);
  holder.phase += dt * 9;
  holder.legs[0].rotation.x = Math.sin(holder.phase) * 0.6;
  holder.legs[1].rotation.x = -Math.sin(holder.phase) * 0.6;
}
